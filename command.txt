# Town Market Commands
# =================
# This file contains all available commands for running and managing the Town Market application.

# Quick Start Commands
# ------------------
# Start both backend and frontend in development mode
bash run.sh dev

# Start both backend and frontend in production mode
bash run.sh prod

# Start Town Market with Docker Compose (dev or prod config)
bash run.sh docker

# Stop all running services (Ctrl+C in the terminal)
# Ctrl+C

# Django Management Commands
# ------------------------
# Run database migrations
python manage.py migrate

# Create superuser
python manage.py createsuperuser

# Run development server
python manage.py runserver

# Run production server with gunicorn
# gunicorn core.wsgi:application

# Run Celery worker
celery -A core worker -l info

# Run Celery beat (scheduler)
celery -A core beat -l info

# Cancel expired orders
python manage.py cancel_expired_orders

# Check system checks
python manage.py check

# Collect static files
python manage.py collectstatic

# Docker Commands
# ------------
# Build and start all Docker services
docker-compose up --build -d

# Stop Docker services
docker-compose down

# View Docker logs
docker-compose logs -f

# Run Django management commands inside Docker
docker-compose exec web python manage.py <command>

# Run migrations inside Docker
docker-compose exec web python manage.py migrate

# Run Town Market in development mode (SQLite)
bash run.sh dev

# Start Town Market in production mode (PostgreSQL)
bash run.sh prod

# Start Town Market with Docker Compose (dev or prod config)
bash run.sh docker

# Connect to PostgreSQL database
# psql -U townmarket -d townmarket -h localhost -p 5432

# Backup database
# pg_dump -U townmarket -d townmarket > backup.sql

# Restore database
# psql -U townmarket -d townmarket < backup.sql


# Cloudflare Services
# -----------------
# Check R2 storage status
python -c "from core.cloudflare import is_r2_configured; print('R2 Configured:', is_r2_configured())"

# Generate R2 URL
python -c "from core.cloudflare import r2; print(r2.url('product_images/logo.png'))"

docker-compose ps
